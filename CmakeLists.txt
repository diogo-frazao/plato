cmake_minimum_required(VERSION 3.25)

set(SDL_STATIC ON)

project("plato" LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Use runtime lib, enable multi processor compilation, enable hot reloading compilation
if (MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_compile_options(/MP)
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "EditAndContinue")
endif()

# Look for thirdParty libs
add_subdirectory(thirdParty/SDL3-3.4.0 EXCLUDE_FROM_ALL)
add_subdirectory(thirdParty/imgui-docking)

add_executable("plato")

# Create RESOURCES_PATH macro and RELEASE_BUILD
if(CMAKE_BUILD_TYPE STREQUAL "Release" OR CMAKE_CONFIGURATION_TYPES STREQUAL "Release")
    target_compile_definitions("plato" PUBLIC RELEASE_BUILD)
    target_compile_definitions("plato" PUBLIC RESOURCES_PATH="./resources")
else()
    target_compile_definitions("plato" PUBLIC RESOURCES_PATH="${CMAKE_CURRENT_SOURCE_DIR}/resources/")
endif()


file(GLOB_RECURSE PLATO_SOURCE CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
file(GLOB_RECURSE PLATO_HEADERS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h")
target_sources("plato" PRIVATE ${PLATO_SOURCE} ${PLATO_HEADERS})

target_include_directories(plato PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")

# Create Visual Studio filters inside project to match folder structure
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}/src" FILES ${PLATO_SOURCE} ${PLATO_HEADERS})

# Link libs
target_link_libraries("plato" PUBLIC SDL3::SDL3-static imgui)
